rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow read access to the artifacts collection for the backend
    match /artifacts/{appId}/public/data/{collection}/{document} {
      allow read: if true; // Allow read access to all documents in this path
      allow write: if request.auth != null; // Only authenticated users can write
    }
    
    // Chat system rules - allow participants to read/write messages
    match /artifacts/{appId}/public/data/chats/{chatId} {
      // Allow read/write if user is a participant in the chat
      allow read, write: if request.auth != null && 
        request.auth.uid in resource.data.participants;
      
      // Allow creation of new chats if user is authenticated
      allow create: if request.auth != null;
      
      // Messages subcollection rules
      match /messages/{messageId} {
        // Allow read if user is a participant in the parent chat
        allow read: if request.auth != null && 
          request.auth.uid in get(/databases/$(database)/documents/artifacts/$(appId)/public/data/chats/$(chatId)).data.participants;
        
        // Allow create (sending messages) if user is a participant
        allow create: if request.auth != null && 
          request.auth.uid in get(/databases/$(database)/documents/artifacts/$(appId)/public/data/chats/$(chatId)).data.participants;
        
        // Allow update/delete only if user is the sender
        allow update, delete: if request.auth != null && 
          request.auth.uid == resource.data.senderId;
      }
    }
    
    // Keep existing rules for other collections
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
